cmake_minimum_required(VERSION 3.20)
project(ArticleCases LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(POLICY CMP0167)
 cmake_policy(SET CMP0167 NEW)
endif()
find_package(MPI REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PETSC REQUIRED IMPORTED_TARGET PETSc)
pkg_check_modules(SLEPC REQUIRED IMPORTED_TARGET SLEPc)
find_package(yaml-cpp REQUIRED)
find_package(OpenMP REQUIRED)
find_package(CGAL REQUIRED)
find_package(TBB REQUIRED)
find_package(VoronoiMeshMaker REQUIRED)
if(NOT TARGET VoronoiMeshMaker::VoronoiMeshMaker)
 find_library(LEGACY_VMM NAMES VoronoiMeshMaker REQUIRED)
 add_library(VoronoiMeshMaker::VoronoiMeshMaker INTERFACE IMPORTED)
 set_target_properties(VoronoiMeshMaker::VoronoiMeshMaker PROPERTIES INTERFACE_LINK_LIBRARIES "${LEGACY_VMM};VoronoiMeshMaker::vmm;TBB::tbb")
endif()
find_package(VoroGradToolkit REQUIRED)
find_library(ADOLC_LIB NAMES adolc REQUIRED)
find_path(ADOLC_INCLUDE NAMES adolc/adolc.h REQUIRED)
get_filename_component(ARTICLE "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
file(GLOB_RECURSE LIB_SOURCES CONFIGURE_DEPENDS "${ARTICLE}/support/library/src/*.cpp")
add_library(diag_historical STATIC ${LIB_SOURCES})
target_include_directories(diag_historical PUBLIC "${ARTICLE}/support/library/include" ${ADOLC_INCLUDE})
target_compile_definitions(diag_historical PUBLIC GSMAKER_USE_SLEPC=1 GSMAKER_USE_ADOLC=1 GSMAKER_USE_VOROGRADTOOLKIT=1 GSMAKER_USE_VORONOI_GRADIENT_KIT=1 GSMAKER_USE_CGAL=1)
target_link_libraries(diag_historical PUBLIC PkgConfig::PETSC PkgConfig::SLEPC MPI::MPI_CXX yaml-cpp::yaml-cpp ${ADOLC_LIB} VoroGradToolkit::VoroGradToolkit VoronoiMeshMaker::VoronoiMeshMaker CGAL::CGAL TBB::tbb OpenMP::OpenMP_CXX)
add_executable(diag_D0 D0_gate/main.cpp)
target_link_libraries(diag_D0 PRIVATE diag_historical)

add_executable(diag_D1 D1_spectral_hypotheses/main.cpp)
target_link_libraries(diag_D1 PRIVATE diag_historical)
